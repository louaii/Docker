docker pull <image_name> //to pull image
docker images //to check the images installed
docker build -t <image_name> . //to build image
docker run -d -p portnb:80 <image_name> //to run the image created
docker ps -a //check containers
docker search httpd //to look for files in library httpd
docker exec -it <container_id_or_name> /bin/bash //to switch to docker container user
docker run -d -p 8080:80 --name container_name img_name
docker tag doc-compose-img:latest <username>/<repo_name>:<name_tag> //to set the image you want to push to dockerhub
docker push <username>/<repo_name>:<tag_name> //to push after setting the image

Dockerfile(httpd):
FROM httpd:latest
COPY ./restoran /usr/local/apache2/htdocs/

Dockerfile(python3onbuild):
FROM python:3-onbuild
COPY . /usr/src/app
CMD ["python", "api.py"]

docker-compose.yaml
//to run docker-compose ==> docker-compose up

yaml file example:
version: '3'
services:
  product-service:
    build: ./product
    volumes:
      - ./product:/usr/src/app
    ports:
      - 5001:80
  website:
    image: php:apache
    volumes:
      - ./website:/var/www/html
    ports:
      - 5000:80
    depends_on:
      - product-service
